-- ========================
-- Priority System State (run order before/after for each function)
-- ========================
-- List of all auto-functions allowed to be prioritized/reordered
-- The number = category order (matches the tab/category number); lower = runs first
local priorityOrder = {
    "1. Main Boss",
    "2. Extra Boss (Kurama/Sukuna)",
    "3. Auto Grab Fruit",
    "4. Mob Farm",
    "5. Auto Skill",
    "6. Train x4",
}
local priorityDropdown = nil

-- GetPriorityRank: returns the order (lower index = first) of a system name, used to compare priority at runtime
local function GetPriorityRank(label)
    for i, v in ipairs(priorityOrder) do
        if v == label then return i end
    end
    return 999
end

-- This system does NOT remove/change the original behavior of each function, it only uses ordering to avoid
-- conflicts (yields) between systems that race to teleport the character in the same frame, e.g. Main Boss vs Mob Farm
-- Returns true = system A (rankA) should get priority over B (rankB) this round
local function PriorityAllows(systemLabel, blockerLabel)
    return GetPriorityRank(systemLabel) <= GetPriorityRank(blockerLabel)
end

local priorityListText = ""
local function RebuildPriorityListText()
    local lines = {}
    for i, v in ipairs(priorityOrder) do
        table.insert(lines, i .. ". " .. v)
    end
    priorityListText = table.concat(lines, "\n")
end
RebuildPriorityListText()

local priorityParagraph = nil
local function RefreshPriorityParagraph()
    RebuildPriorityListText()
    if priorityParagraph then
        pcall(function() priorityParagraph:SetDesc(priorityListText) end)
    end
end

-- The real save-to-file function is defined further below the script (near the Auto Save/Load section)
-- It's forward-declared here so movePriority() can call it immediately without reordering the whole file
local savePriorityOrderToFile = function() end

local function movePriority(label, direction)
    local idx = nil
    for i, v in ipairs(priorityOrder) do
        if v == label then idx = i break end
    end
    if not idx then return end
    local newIdx = idx + direction
    if newIdx < 1 or newIdx > #priorityOrder then return end
    priorityOrder[idx], priorityOrder[newIdx] = priorityOrder[newIdx], priorityOrder[idx]
    RefreshPriorityParagraph()
    savePriorityOrderToFile()
end

local function findProximityPrompt(model)
    if not model then return nil end
    for _, v in ipairs(model:GetDescendants()) do
        if v:IsA("ProximityPrompt") then return v end
    end
    return nil
end

local function triggerPrompt(prompt)
    if not prompt then return false end
    pcall(function()
        if fireproximityprompt then
            fireproximityprompt(prompt)
        else
            pressKey(prompt.KeyboardKeyCode ~= Enum.KeyCode.Unknown and prompt.KeyboardKeyCode or Enum.KeyCode.E)
        end
    end)
    return true
end

local function getBossPosition(bossName)
    local folderName = bossPaths[bossName]
    if not folderName then return nil end
    local clientEntities = workspace:FindFirstChild("ClientEntities")
    if not clientEntities then return nil end
    local bossModel = clientEntities:FindFirstChild(folderName)
    if not bossModel then return nil end
    local root = bossModel:FindFirstChild("HumanoidRootPart") or bossModel:FindFirstChild("RootPart") or bossModel.PrimaryPart
    if not root then
        for _, part in ipairs(bossModel:GetDescendants()) do
            if part:IsA("BasePart") then root = part break end
        end
    end
    return root and root.Position
end

local function checkAnyBossActive()
    for bossName, enabled in pairs(targetBosses) do
        if enabled and getBossPosition(bossName) then
            return true
        end
    end
    return false
end

-- ========================
-- Slot Stroke Engine
-- ========================
local function getSlotStroke()
    local ok, stroke = pcall(function()
        local playerGui = LocalPlayer:FindFirstChild("PlayerGui")
        local hotbar    = playerGui and playerGui:FindFirstChild("Main") and playerGui.Main:FindFirstChild("HUD") and playerGui.Main.HUD:FindFirstChild("Hotbar")
        if not hotbar then return nil end
        local slot = nil
        local slotName = hotbarSlotNames[selectedSlot]
        if slotName then slot = hotbar:FindFirstChild(slotName) end
        if not slot then
            for _, child in ipairs(hotbar:GetChildren()) do
                if child:IsA("GuiObject") and child.LayoutOrder == selectedSlot then
                    slot = child break
                end
            end
        end
        if not slot then
            local validSlots = {}
            for _, child in ipairs(hotbar:GetChildren()) do
                if child:IsA("ImageButton") or child:IsA("TextButton") or child.Name:lower():find("slot") then
                    table.insert(validSlots, child)
                end
            end
            table.sort(validSlots, function(a, b) return (a.LayoutOrder or 0) < (b.LayoutOrder or 0) end)
            slot = validSlots[selectedSlot]
        end
        return slot and slot:FindFirstChildOfClass("UIStroke") or nil
    end)
    return ok and stroke or nil
end

local function isStrokeBlack(stroke)
    if not stroke then return false end
    local c = stroke.Color
    return c.R < 0.1 and c.G < 0.1 and c.B < 0.1
end

-- Delay after teleporting to a Fruit model before pressing ProximityPrompt: changed from 3s -> 1.5s as requested
local FRUIT_GRAB_DELAY = 1.5

local function tryGrabFruit(fruitName)
    local model = workspace:FindFirstChild(fruitName)
    if not model then return false end
    local hrp = getHRP()
    if not hrp then return false end

    -- Save the original position before going to grab
    local originalPosition = hrp.CFrame
    
    local targetPart = model.PrimaryPart or model:FindFirstChild("HumanoidRootPart") or model:FindFirstChild("RootPart")
    if not targetPart then
        for _, part in ipairs(model:GetDescendants()) do
            if part:IsA("BasePart") then targetPart = part break end
        end
    end
    if not targetPart then return false end

    hrp.CFrame = CFrame.new(targetPart.Position + Vector3.new(0, 3, 0))
    WindUI:Notify({ Title = "Found: " .. fruitName, Content = "Teleported! Waiting " .. FRUIT_GRAB_DELAY .. "s before grabbing...", Duration = 3, Icon = "zap" })
    task.wait(FRUIT_GRAB_DELAY)

    local stillExists = workspace:FindFirstChild(fruitName)
    if not stillExists then
        WindUI:Notify({ Title = fruitName .. " disappeared!", Duration = 2, Icon = "x" })
        -- Return to the original position even if the fruit disappeared
        hrp.CFrame = originalPosition
        return false
    end

    local prompt = findProximityPrompt(stillExists)
    if prompt then
        triggerPrompt(prompt)
        WindUI:Notify({ Title = "Grabbed: " .. fruitName, Content = "ProximityPrompt triggered! Returning to position...", Duration = 3, Icon = "check-circle" })
        task.wait(1)
        -- Return to the original position after a successful grab
        hrp.CFrame = originalPosition
        return true
    else
        WindUI:Notify({ Title = "No Prompt on " .. fruitName, Duration = 2, Icon = "alert-circle" })
        -- Return to the original position if no Prompt is found
        hrp.CFrame = originalPosition
        return false
    end
end

        if fruit.enabled then
            local model = workspace:FindFirstChild(fruit.name)

                task.wait(1)

        end
    end
end

local function goToNpc(npcName)
    if not npcName or npcName == "" then
        WindUI:Notify({ Title = "Go to NPC", Content = "Please select an NPC first!", Duration = 3, Icon = "alert-circle" })
        return
    end
    
    local questsFolder = workspace:FindFirstChild("Quests")
    if not questsFolder then return end
    
    local npc = questsFolder:FindFirstChild(npcName)
    if not npc then
        WindUI:Notify({ Title = "NPC Not Found", Content = npcName .. " not found!", Duration = 3, Icon = "x" })
        return
    end
    
    local targetPart = npc.PrimaryPart or npc:FindFirstChild("HumanoidRootPart") or npc:FindFirstChild("RootPart")
    if targetPart then
        local hrp = getHRP()
        if hrp then
            hrp.CFrame = CFrame.new(targetPart.Position + Vector3.new(3, 0, 3))
            WindUI:Notify({ Title = "Teleported to NPC", Content = "Arrived at " .. npcName, Duration = 3, Icon = "map-pin" })
        end
    else
        WindUI:Notify({ Title = "Invalid NPC", Content = "No valid part found on " .. npcName, Duration = 3, Icon = "alert-circle" })
    end
end

-- ========================
-- Main Boss Farm Engine (Priority 1)
-- ========================
-- Changed from teleporting straight into the boss model to: hover 30 studs above the boss's head,
-- face down at the model, and continuously left-click on the boss while in range. This avoids the old
-- bug where standing inside hitboxes during skill release could glitch/cancel animations.
local BOSS_HOVER_HEIGHT = 30
local bossHoverConn       = nil
local bossHoverTargetName = nil
local bossMouseClickThread = nil

local function setBossHRPAnchored(state)
    local hrp = getHRP()
    if hrp then pcall(function() hrp.Anchored = state end) end
end

local function stopBossMouseClick()
    if bossMouseClickThread then task.cancel(bossMouseClickThread) bossMouseClickThread = nil end
end

local function startBossMouseClick()
    stopBossMouseClick()
    bossMouseClickThread = task.spawn(function()
        while bossHoverTargetName do
            local bossPos = getBossPosition(bossHoverTargetName)
            if bossPos then
                pcall(function()
                    local screenPos, onScreen = Camera:WorldToViewportPoint(bossPos)
                    if onScreen then
                        VirtualInputManager:SendMouseButtonEvent(screenPos.X, screenPos.Y, 0, true, game, 1)
                        task.wait(0.05)
                        VirtualInputManager:SendMouseButtonEvent(screenPos.X, screenPos.Y, 0, false, game, 1)
                    end
                end)
            end
            task.wait(0.15)
        end
    end)
end

local function stopBossHoverLock()
    if bossHoverConn then bossHoverConn:Disconnect() bossHoverConn = nil end
    bossHoverTargetName = nil
    stopBossMouseClick()
    setBossHRPAnchored(false)
end

local function startBossHoverLock(bossName)
    bossHoverTargetName = bossName
    if bossHoverConn then return end -- already running, just retarget (set above)
    setBossHRPAnchored(true)
    bossHoverConn = RunService.Heartbeat:Connect(function()
        if not bossHoverTargetName then return end
        local hrp = getHRP()
        if not hrp then return end
        local bossPos = getBossPosition(bossHoverTargetName)
        if not bossPos then return end
        local hoverPos = bossPos + Vector3.new(0, BOSS_HOVER_HEIGHT, 0)
        -- The boss is directly below the hover point, which is a degenerate case for CFrame.new(pos, lookAt)
        -- (the look direction would be parallel to the default up vector and produce undefined/NaN rotation).
        -- Use an explicit rotation instead: tilt -90 degrees on X so the character faces straight down at it.
        hrp.CFrame = CFrame.new(hoverPos) * CFrame.Angles(math.rad(-90), 0, 0)
    end)
    startBossMouseClick()
end

-- On respawn (e.g. after Auto Reset death), the old HRP reference is gone; if a boss farm is still meant
-- to be running, simply restart the lock fresh on the new character instead of leaving it stuck mid-state
local onCharacterRespawnedForBossLock = function()
    if bossHoverConn then
        local resumeBoss = bossHoverTargetName
        bossHoverConn:Disconnect()
        bossHoverConn = nil
        stopBossMouseClick()
        if resumeBoss and farmingEnabled then
            startBossHoverLock(resumeBoss)
        end
    end
end

LocalPlayer.CharacterAdded:Connect(function()
    task.wait(2)
    onCharacterRespawnedForBossLock()
end)

local function updateFarmState()
    local anyBossEnabled = false
    for _, enabled in pairs(targetBosses) do
        if enabled then anyBossEnabled = true break end
    end

    if not anyBossEnabled then
        farmingEnabled = false
        isBossActiveNow = false
        stopBossHoverLock()
        if farmThread then task.cancel(farmThread) farmThread = nil end
        return
    end

    if farmingEnabled then return end
    farmingEnabled = true

    farmThread = task.spawn(function()
        while farmingEnabled do
            if trainX4Enabled then
                stopBossHoverLock()
                task.wait(0.5)
                continue
            end

            local hasTarget = false
            local activeBossCount = 0
            local currentBossName = nil

            for bossName, enabled in pairs(targetBosses) do
                if not enabled then continue end
                activeBossCount = activeBossCount + 1
                local bossPos = getBossPosition(bossName)
                if bossPos then
                    isBossActiveNow = true
                    currentBossName = bossName
                    hasTarget = true
                    hasResetThisCycle = false
                    break -- hover-lock on the first active target found; the Heartbeat lock re-reads its live position every frame
                end
            end

            if hasTarget and currentBossName then
                startBossHoverLock(currentBossName)
                task.wait(0.2)
            end

            if not hasTarget then
                isBossActiveNow = false
                stopBossHoverLock()
                if activeBossCount > 0 then
                    if autoResetEnabled and not hasResetThisCycle then
                        local char = LocalPlayer.Character
                        local hum = char and char:FindFirstChildOfClass("Humanoid")
                        if hum and hum.Health > 0 then
                            hasResetThisCycle = true
                            hum.Health = 0
                            task.wait(0.5)
                        end
                    end
                    if useSavePosEnabled and savedPosition then
                        local hrp = getHRP()
                        if hrp then hrp.CFrame = CFrame.new(savedPosition) task.wait(0.3) end
                    end
                    task.wait(0.3)
                else
                    task.wait(0.1)
                end
            else
                task.wait(0.1)
            end
        end
        stopBossHoverLock()
    end)
end

    autoSkillThread = task.spawn(function()

                if isEquipping then task.wait(0.1) continue end
                local firedAny = false
                for keyStr, isSelected in pairs(selectedSkills) do

                            task.spawn(function() pressKey(Enum.KeyCode[keyStr]) end)
                        end
                    end
                end
                task.wait(firedAny and 0.3 or 0.1)
            else
                task.wait(0.5)
            end
        end
    end)
end

-- ========================
-- Mob Farm Engine (Priority 4)
-- Fixed bug: Auto Mob not working:
--   Original cause: the Toggle set farmMobEnabled directly, then called updateMobFarmState()
--   but updateMobFarmState() checked "if not farmMobEnabled then ... return end" right at the top
--   If the flag value hadn't updated yet (e.g. loaded from config before the UI was ready) or was
--   called twice in a row, the old thread reference would be left dangling (not cancelled) while a new
--   thread spawned on top of it, leaving farmMobThread pointing at the wrong thread. Toggling it off/on
--   repeatedly would desync the state and stop teleporting to mobs entirely.
--   Fix: make updateMobFarmState() safe to call repeatedly (idempotent) by always stopping the old one
--   before starting a new one, and add error checks in case mobName no longer exists before teleporting.
-- ========================
local function scanClientEntities()
    local mobsFound = {}
    local clientEntities = workspace:FindFirstChild("ClientEntities")
    if clientEntities then
        for _, child in ipairs(clientEntities:GetChildren()) do
            if child:IsA("Model") and not table.find(mobsFound, child.Name) and not bossPaths[child.Name] and child.Name ~= "Kurama" and child.Name ~= "CurseKing" then
                table.insert(mobsFound, child.Name)
            end
        end
    end
    if #mobsFound == 0 then table.insert(mobsFound, "No Mobs Found") end
    table.sort(mobsFound)
    availableMobs = mobsFound
    return mobsFound
end

local function getMobModel(mobName)

-- Hover ~2 studs above the mob model's head, and force the character to face down (look down) at the
-- monster model below, locked firmly in place every frame (anchored) so it never falls or drifts before
-- snapping back up. This replaces the old "set CFrame once every 0.4s" approach, which let gravity pull the
-- character down in between updates (since PlatformStand alone does not disable gravity) — that was the
-- cause of the "floats then drops" look the user reported.
local MOB_HOVER_HEIGHT = 2
local mobHoverConn = nil
local mobHoverTargetPos = nil

local function setHRPAnchored(state)
    local hrp = getHRP()
    if hrp then pcall(function() hrp.Anchored = state end) end
end

local function startMobHoverLock()
    if mobHoverConn then return end -- already running, avoid double connections
    setHRPAnchored(true)
    mobHoverConn = RunService.Heartbeat:Connect(function()
        if not mobHoverTargetPos then return end
        local hrp = getHRP()
        if not hrp then return end
        local hoverPos = mobHoverTargetPos + Vector3.new(0, MOB_HOVER_HEIGHT, 0)
        -- The monster is directly below the hover point, which is a degenerate case for CFrame.new(pos, lookAt)
        -- (the look direction is parallel to the default up vector and produces undefined/NaN rotation).
        -- Use an explicit rotation instead: tilt the character -90 degrees on X so it faces straight down.
        hrp.CFrame = CFrame.new(hoverPos) * CFrame.Angles(math.rad(-90), 0, 0)
    end)
end

local function stopMobHoverLock()
    if mobHoverConn then mobHoverConn:Disconnect() mobHoverConn = nil end
    mobHoverTargetPos = nil
    setHRPAnchored(false)
end

-- On respawn (e.g. after Auto Reset death), the old HRP reference is gone; if the mob farm is still meant
-- to be running, simply restart the lock fresh on the new character instead of leaving it stuck mid-state
onCharacterRespawnedForMobLock = function()
    if mobHoverConn then
        mobHoverConn:Disconnect()
        mobHoverConn = nil
        if farmMobEnabled and mobHoverTargetPos then
            startMobHoverLock()
        end
    end
end

local function applyProneOverMob(hrp, mobPos)
    if not hrp or not mobPos then return end
    mobHoverTargetPos = mobPos
    startMobHoverLock()

    local char = getCharacter()
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    if hum then
        pcall(function() hum.PlatformStand = true end)
    end
end

local function releaseProneState()
    stopMobHoverLock()
    local char = getCharacter()
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    if hum then
        pcall(function() hum.PlatformStand = false end)
    end
end

local function updateMobFarmState()
    -- Bug fix: always stop the old thread before starting a new one, so there are never two threads overlapping
    if farmMobThread then
        pcall(task.cancel, farmMobThread)
        farmMobThread = nil
    end

    if not farmMobEnabled then
        releaseProneState()
        return
    end

    farmMobThread = task.spawn(function()
        while farmMobEnabled do
            if checkAnyBossActive() or isKuramaInArena or isSukunaInArena or getAnyPendingFruitModel() or trainX4Enabled then
                releaseProneState()
                task.wait(0.5)
                continue
            end

            local anySelected = false
            for _, isSelected in pairs(selectedMobs) do
                if isSelected then anySelected = true break end
            end

            if not anySelected then
                releaseProneState()
                task.wait(0.5)
                continue
            end

            local teleportsDone = false
            for mobName, isSelected in pairs(selectedMobs) do
                if not farmMobEnabled then break end
                if isSelected and mobName ~= "No Mobs Found" then
                    local mobModel = getMobModel(mobName)
                    local pos = getMobRoot(mobModel) and getMobRoot(mobModel).Position or nil
                    local hrp = getHRP()
                    if hrp and pos then
                        applyProneOverMob(hrp, pos)
                        teleportsDone = true
                        task.wait(0.4)
                    end
                end
            end

            if not teleportsDone then
                releaseProneState()
                task.wait(0.5)
            else
                task.wait(0.1)
            end
        end
        releaseProneState()
    end)
end

-- ========================
-- Server Hop Logic (scans only for servers aged 30+ minutes)
-- ========================
local function fetchServerList(maxPages)
    local all = {}
    local cursor = nil
    for _ = 1, (maxPages or 3) do
        local url = "https://games.roblox.com/v1/games/" .. game.PlaceId .. "/servers/Public?sortOrder=Asc&limit=100" .. (cursor and ("&cursor=" .. cursor) or "")
        local ok, raw = pcall(function() return game:HttpGet(url) end)
        if not ok or not raw then return all, false end

        local ok2, decoded = pcall(function() return HttpService:JSONDecode(raw) end)
        if not ok2 or not decoded or not decoded.data then return all, false end

        for _, server in ipairs(decoded.data) do
            table.insert(all, server)
        end

        cursor = decoded.nextPageCursor
        if not cursor or cursor == "" then break end
    end
    return all, true
end

local function serverHop()
    WindUI:Notify({ Title = "Server Hop", Content = "Scanning for servers older than " .. tostring(hopMinAge) .. " minutes...", Duration = 4, Icon = "refresh-cw" })

    local servers, fetchOk = fetchServerList(3)

    if not fetchOk or #servers == 0 then
        WindUI:Notify({ Title = "Server Hop Failed", Content = "Could not fetch server list (HTTP/Roblox API issue). Please try again.", Duration = 5, Icon = "alert-triangle" })
        return
    end

    -- Note: the Roblox Public Servers API has no direct "server age" value
    -- so we sort Asc (old -> new) and skip the most-recent portion based on hopMinAge
    -- to reduce the chance of hopping into a freshly-opened server
    local skipRatio = math.clamp((hopMinAge or 30) / 60, 0, 0.85)
    local skipCount = math.floor(#servers * skipRatio)

    local candidates = {}
    for i, server in ipairs(servers) do
        if i > skipCount and server.id ~= game.JobId and server.playing and server.maxPlayers and server.playing < server.maxPlayers then
            table.insert(candidates, server)
        end
    end

    -- Safety net: if no server passes the age condition, fall back to any available server
    if #candidates == 0 then
        for _, server in ipairs(servers) do
            if server.id ~= game.JobId and server.playing and server.maxPlayers and server.playing < server.maxPlayers then
                table.insert(candidates, server)
            end
        end
    end

    if #candidates > 0 then
        local target = candidates[math.random(1, #candidates)]
        local teleportOk, teleportErr = pcall(function()
            TeleportService:TeleportToPlaceInstance(game.PlaceId, target.id, LocalPlayer)
        end)
        if not teleportOk then
            WindUI:Notify({ Title = "Server Hop Failed", Content = "Teleport failed: " .. tostring(teleportErr), Duration = 5, Icon = "alert-triangle" })
        end
    else
        WindUI:Notify({ Title = "Server Hop", Content = "No other available server found right now. Please try again later.", Duration = 4, Icon = "alert-triangle" })
    end
end

local function stopAutoHop()
    autoHopEnabled = false
    if autoHopThread then
        pcall(task.cancel, autoHopThread)
        autoHopThread = nil
    end
end

local function startAutoHop()
    stopAutoHop()
    autoHopEnabled = true
    autoHopThread = task.spawn(function()
        while autoHopEnabled do
            local ok, err = pcall(serverHop)
            if not ok then
                warn("[Divinex] AutoHop error: " .. tostring(err))
                WindUI:Notify({ Title = "Auto Hop", Content = "An error occurred: " .. tostring(err), Duration = 4, Icon = "alert-triangle" })
            end
            if not autoHopEnabled then break end
            local waitTime = (hopInterval and hopInterval > 0) and (hopInterval * 60) or 600
            task.wait(waitTime)
        end
    end)
end

-- ========================
-- 10. Render Settings (extend the game's limited Mob/Boss view distance)
-- ========================
-- The game limits how far away Mob/Boss models stream in via workspace.StreamingTargetRadius /
-- StreamingMinRadius. These are officially "NotScriptable" properties in normal scripts, but most
-- executors expose sethiddenproperty/set_scriptable to write them anyway. Every write below is wrapped
-- in pcall and tries multiple known executor function names, falling back gracefully if none exist —
-- so this never throws even on executors that don't support it.
local renderStreamingRadius = 4096
local renderMinRadius       = 1024
local renderSettingsApplied = false

local function trySetWorkspaceProperty(propName, value)
    -- Attempt 1: direct write (works on some executors where the property isn't actually locked)
    local ok1 = pcall(function() workspace[propName] = value end)
    if ok1 and workspace[propName] == value then return true end

    -- Attempt 2: sethiddenproperty (Synapse/older executor convention)
    if sethiddenproperty then
        local ok2 = pcall(function() sethiddenproperty(workspace, propName, value) end)
        if ok2 then return true end
    end

    -- Attempt 3: set_scriptable / setscriptable then direct write (newer executor convention)
    local setScriptableFn = set_scriptable or setscriptable
    if setScriptableFn then
        local ok3 = pcall(function()
            setScriptableFn(workspace, propName, true)
            workspace[propName] = value
        end)
        if ok3 then return true end
    end

    return false
end

local function applyRenderSettings()
    local okRadius = trySetWorkspaceProperty("StreamingTargetRadius", renderStreamingRadius)
    local okMin     = trySetWorkspaceProperty("StreamingMinRadius", renderMinRadius)

    -- Also push the player's camera zoom cap out, since a short max zoom can make distant Mob/Boss
    -- models harder to see/click even once they're streamed in
    pcall(function() LocalPlayer.CameraMaxZoomDistance = math.max(LocalPlayer.CameraMaxZoomDistance, 400) end)

    renderSettingsApplied = okRadius or okMin
    return okRadius, okMin
end

local function resetRenderSettings()
    trySetWorkspaceProperty("StreamingTargetRadius", 1024) -- Roblox's documented default
    trySetWorkspaceProperty("StreamingMinRadius", 64)       -- Roblox's documented default
    pcall(function() LocalPlayer.CameraMaxZoomDistance = 100 end)
    renderSettingsApplied = false
end

local dragging, dragInput, dragStart, startPos
FloatBtn.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragStart = input.Position
        startPos = FloatBtn.Position
        input.Changed:Connect(function() if input.UserInputState == Enum.UserInputState.End then dragging = false end end)
    end
end)
FloatBtn.InputChanged:Connect(function(input) if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then dragInput = input end end)
UserInputService.InputChanged:Connect(function(input)
    if dragging and input == dragInput then
        local delta = input.Position - dragStart
        FloatBtn.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
    end
end)

FloatBtn.MouseButton1Click:Connect(function()
    pcall(function()
        if Window.Toggle then Window:Toggle()
        elseif Window.Show then Window:Show() end
    end)
end)

task.spawn(function()
    while true do
        task.wait(0.5)
        pcall(function()
            local currentUiVisible = false
            for _, gui in ipairs(coreGui:GetChildren()) do
                if gui.Name == "WindUI" and gui:IsA("ScreenGui") then currentUiVisible = gui.Enabled break end
            end
            FloatBtn.Visible = not currentUiVisible
        end)
    end
end)

        selectedSlot = tonumber(loadedSlot:match("%d")) or 1
        if GetValue("autoEquip", false) then startAutoEquip() end

        -- 2. Extra Boss Farm (Kurama + Sukuna)
        local loadedSkills = GetValue("selectedSkills", {})
        local newSkills = {}
        if type(loadedSkills) == "table" then for _, k in pairs(loadedSkills) do newSkills[k] = true end
        elseif type(loadedSkills) == "string" then newSkills[loadedSkills] = true end
        selectedSkills = newSkills
        if GetValue("autoSkill", false) then startAutoSkill() end

        local loadedKuramaSkills = GetValue("kuramaSelectedSkills", {})
        local newKuramaSkills = {}
        if type(loadedKuramaSkills) == "table" then for _, k in pairs(loadedKuramaSkills) do newKuramaSkills[k] = true end
        elseif type(loadedKuramaSkills) == "string" then newKuramaSkills[loadedKuramaSkills] = true end
        kuramaSelectedSkills = newKuramaSkills

        local loadedSukunaSkills = GetValue("sukunaSelectedSkills", {})
        local newSukunaSkills = {}
        if type(loadedSukunaSkills) == "table" then for _, k in pairs(loadedSukunaSkills) do newSukunaSkills[k] = true end
        elseif type(loadedSukunaSkills) == "string" then newSukunaSkills[loadedSukunaSkills] = true end
        sukunaSelectedSkills = newSukunaSkills

        autoKuramaEnabled = GetValue("enableKurama", false)
        if autoKuramaEnabled then startAutoKurama() end

        autoSukunaEnabled = GetValue("enableSukuna", false)
        if autoSukunaEnabled then startAutoSukuna() end

        -- 3. Fruit
        for i, fruit in ipairs(fruitTargets) do fruitTargets[i].enabled = GetValue("fruit_" .. fruit.name, false) end
        if GetValue("autoGrab", false) then startAutoGrab() end

        -- 4. Mob Farm
        local loadedMobs = GetValue("selectedMobsFlag", {})
        local newMobs = {}
        if type(loadedMobs) == "table" then for _, m in pairs(loadedMobs) do newMobs[m] = true end
        elseif type(loadedMobs) == "string" then newMobs[loadedMobs] = true end
        selectedMobs = newMobs

        farmMobEnabled = GetValue("enableMobFarm", false)
        updateMobFarmState()

        -- 6. Train x4 (not auto-started for CFrame-lock safety on fresh map load, but the flag is kept)
        -- Note: the user can turn it on manually right after the scene loads

        -- 7. Teleport (Zone / NPC selection)
        local loadedZone = GetValue("teleportZone", nil)
        if loadedZone and loadedZone ~= "" then selectedZone = loadedZone end
        local loadedNpc = GetValue("selectedNpc", nil)
        if loadedNpc and loadedNpc ~= "" then selectedNpc = loadedNpc end

        -- 8. Priority order
        -- (Already loaded from its own priority_order.json file at script start via loadPriorityOrderFromFile())

        -- 9. Other (AntiAfk / Hop)
        if GetValue("antiAfk", true) then startAntiAfk() end
        local loadedInterval = GetValue("hopInterval", 10)
        hopInterval = tonumber(loadedInterval) or 10
        if GetValue("autoHop", false) then startAutoHop() end

        -- 10. Render Settings
        renderStreamingRadius = tonumber(GetValue("renderStreamRadius", GetValue("renderStreamRadiusInput", 4096))) or 4096
        renderMinRadius       = tonumber(GetValue("renderMinRadius", GetValue("renderMinRadiusInput", 1024))) or 1024
        if GetValue("renderAutoApply", false) then applyRenderSettings() end
    end)
end)

-- Persist the current priority order to its own file (doesn't rely on a non-existent WindUI API)
-- Uses the executor's writefile/readfile directly, which is safer and more reliable
local PRIORITY_FILE = "DivinexConfigs/priority_order.json"

savePriorityOrderToFile = function()
    pcall(function()
        if writefile then
            if makefolder and not isfolder("DivinexConfigs") then makefolder("DivinexConfigs") end
            writefile(PRIORITY_FILE, HttpService:JSONEncode(priorityOrder))
        end
    end)
end

local function loadPriorityOrderFromFile()
    local ok, result = pcall(function()
        if readfile and isfile and isfile(PRIORITY_FILE) then
            local raw = readfile(PRIORITY_FILE)
            local decoded = HttpService:JSONDecode(raw)
            if type(decoded) == "table" and #decoded == #priorityOrder then
                return decoded
            end
        end
        return nil
    end)
    if ok and result then
        priorityOrder = result
        RefreshPriorityParagraph()
    end
end

loadPriorityOrderFromFile()

task.spawn(function()
    while true do
        task.wait(5)
        savePriorityOrderToFile()
    end
end)
