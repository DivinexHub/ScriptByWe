-- ========================
-- Helpers

local function getCharacterModel(player)
    return player and player.Character
end

local function getHead(player)
    local char = getCharacterModel(player)
    return char and char:FindFirstChild("Head")
end

-- ========================
-- ESP System
-- ========================
local ESP_COLORS = {
    murder  = Color3.fromRGB(255, 50, 50),
    sheriff = Color3.fromRGB(50, 120, 255),
    default = Color3.fromRGB(200, 200, 200),
}

local espGui = Instance.new("ScreenGui")
espGui.Name = "AAS_ESP"
espGui.ResetOnSpawn = false
pcall(function() espGui.Parent = game:GetService("CoreGui") end)

local espHighlights = {}

local function removeESP(player)
    if espHighlights[player.Name] then
        pcall(function() espHighlights[player.Name]:Destroy() end)
        espHighlights[player.Name] = nil
    end
end

local function applyESP(player)
    if player == LocalPlayer then return end
    removeESP(player)

    local char = player.Character
    if not char then return end

    -- Use Highlight (Roblox highlight object, works client-side in executor)
    local highlight = Instance.new("Highlight")
    highlight.Name = "AAS_Highlight_" .. player.Name
    highlight.FillTransparency = 0.5
    highlight.OutlineTransparency = 0

    local color = ESP_COLORS.default
    if detectedMurderer == player.Name then
        color = ESP_COLORS.murder
    elseif detectedSheriff == player.Name then
        color = ESP_COLORS.sheriff
    end

    highlight.FillColor    = color
    highlight.OutlineColor = color
    highlight.Adornee      = char
    highlight.Parent       = espGui
    espHighlights[player.Name] = highlight
end

local function refreshAllESP()
    if not espEnabled then
        for _, h in pairs(espHighlights) do pcall(function() h:Destroy() end) end
        espHighlights = {}
        return
    end
    for _, p in ipairs(Players:GetPlayers()) do
        applyESP(p)
    end
end

local function startESP()
    espEnabled = true
    refreshAllESP()

    if espThread then task.cancel(espThread) end
    espThread = task.spawn(function()
        while espEnabled do
            for _, p in ipairs(Players:GetPlayers()) do
                if p ~= LocalPlayer then
                    if not espHighlights[p.Name] or not espHighlights[p.Name].Parent then
                        applyESP(p)
                    end
                    -- Update color when role changes
                    if espHighlights[p.Name] then
                        local color = ESP_COLORS.default
                        if detectedMurderer == p.Name then
                            color = ESP_COLORS.murder
                        elseif detectedSheriff == p.Name then
                            color = ESP_COLORS.sheriff
                        end
                        pcall(function()
                            espHighlights[p.Name].FillColor    = color
                            espHighlights[p.Name].OutlineColor = color
                        end)
                    end
                end
            end
            task.wait(0.5)
        end
    end)

    -- Handle new players joining
    table.insert(espConnections, Players.PlayerAdded:Connect(function(p)
        task.wait(2)
        if espEnabled then applyESP(p) end
    end))
    -- Handle players leaving
    table.insert(espConnections, Players.PlayerRemoving:Connect(function(p)
        removeESP(p)
    end))
    -- Handle character respawn
    table.insert(espConnections, Players.PlayerAdded:Connect(function(p)
        p.CharacterAdded:Connect(function()
            task.wait(1)
            if espEnabled then applyESP(p) end
        end)
    end))
    for _, p in ipairs(Players:GetPlayers()) do
        p.CharacterAdded:Connect(function()
            task.wait(1)
            if espEnabled then applyESP(p) end
        end)
    end
end

local function stopESP()
    espEnabled = false
    for _, conn in ipairs(espConnections) do pcall(function() conn:Disconnect() end) end
    espConnections = {}
    if espThread then task.cancel(espThread) espThread = nil end
    for _, h in pairs(espHighlights) do pcall(function() h:Destroy() end) end
    espHighlights = {}
end

-- ========================
-- Role Label System (BillboardGui floating above head)
-- ========================
local labelGui = Instance.new("ScreenGui")
labelGui.Name = "AAS_RoleLabels"
labelGui.ResetOnSpawn = false
pcall(function() labelGui.Parent = game:GetService("CoreGui") end)

local function removeRoleLabel(playerName)
    if roleLabels[playerName] then
        pcall(function() roleLabels[playerName]:Destroy() end)
        roleLabels[playerName] = nil
    end
end

local function setRoleLabel(player, roleType)
    -- roleType = "murder" | "sheriff" | nil (clear)
    local head = getHead(player)
    if not head then
        -- Wait briefly in case character is loading
        task.spawn(function()
            task.wait(1)
            head = getHead(player)
            if head then setRoleLabel(player, roleType) end
        end)
        return
    end

    removeRoleLabel(player.Name)
    if not roleType then return end

    local billboard = Instance.new("BillboardGui")
    billboard.Name         = "AAS_Role_" .. player.Name
    billboard.Size         = UDim2.fromOffset(160, 50)
    billboard.StudsOffset  = Vector3.new(0, 3.5, 0)
    billboard.AlwaysOnTop  = true
    billboard.Adornee      = head
    billboard.Parent       = labelGui

    local label = Instance.new("TextLabel")
    label.Size              = UDim2.fromScale(1, 1)
    label.BackgroundTransparency = 1
    label.TextScaled        = true
    label.Font              = Enum.Font.GothamBold
    label.TextStrokeTransparency = 0
    label.TextStrokeColor3  = Color3.fromRGB(0, 0, 0)

    if roleType == "murder" then
        label.Text       = "⚠ MURDER"
        label.TextColor3 = Color3.fromRGB(255, 30, 30)
    elseif roleType == "sheriff" then
        label.Text       = "🔵 SHERIFF"
        label.TextColor3 = Color3.fromRGB(50, 140, 255)
    end

    label.Parent = billboard
    roleLabels[player.Name] = billboard
end

local function clearAllRoleLabels()
    for name, _ in pairs(roleLabels) do
        removeRoleLabel(name)
    end
    roleLabels = {}
    detectedMurderer = nil
    detectedSheriff  = nil
end

-- ========================
-- Role Detection Logic
-- ========================
-- Strategy:
-- Murder: scan workspace.Akura_Gameza (model with that prefix, not always identical name)
--         OR scan each player's character for a "Knife" child
-- Sheriff: scan each player's character for a "Gun" child
--
-- We use FindFirstChild on the character because the child name "Knife"/"Gun" is the reliable signal,
-- and the player's character folder name is their display/username.

local function scanRoles()
    local newMurder  = nil
    local newSheriff = nil

    -- Method 1: Check workspace for Akura_Gameza (murderer indicator model)
    -- The model name may vary, so scan for any child starting with "Akura_Gameza"
    local wsChildren = workspace:GetChildren()
    for _, obj in ipairs(wsChildren) do
        if obj.Name:find("Akura_Gameza") then
            -- Try to match this model to a player by checking who's nearby or by
            -- checking if any player's character has a part with the same name
            -- Heuristic: find the player whose character name appears in this model
            for _, p in ipairs(Players:GetPlayers()) do
                if p ~= LocalPlayer then
                    local char = p.Character
                    if char and char:FindFirstChild("Akura_Gameza", true) then
                        newMurder = p.Name
                        break
                    end
                end
            end
            -- If still not matched, flag whoever owns the model (best effort)
            if not newMurder then
                local humanoid = obj:FindFirstChildOfClass("Humanoid")
                if humanoid then
                    for _, p in ipairs(Players:GetPlayers()) do
                        if p.Character and p.Character.Name == obj.Name then
                            newMurder = p.Name break
                        end
                    end
                end
            end
        end
    end

    -- Method 2: Scan each player's character for Knife / Gun child
    for _, p in ipairs(Players:GetPlayers()) do
        if p == LocalPlayer then continue end
        local char = p.Character
        if not char then continue end

        -- Check for Knife anywhere under the character (including tools held)
        if char:FindFirstChild("Knife", true) then
            newMurder = p.Name
        end
        -- Check for Gun anywhere under the character
        if char:FindFirstChild("Gun", true) then
            newSheriff = p.Name
        end
    end

    return newMurder, newSheriff
end

local function startRoleDetect()
    roleDetectEnabled = true
    if roleThread then task.cancel(roleThread) end
    roleThread = task.spawn(function()
        while roleDetectEnabled do
            local ok, murder, sheriff = pcall(scanRoles)
            if ok then
                -- Murderer changed
                if murder ~= detectedMurderer then
                    if detectedMurderer then
                        local oldPlayer = Players:FindFirstChild(detectedMurderer)
                        if oldPlayer then setRoleLabel(oldPlayer, nil) end
                        if espEnabled and oldPlayer then applyESP(oldPlayer) end
                    end
                    detectedMurderer = murder
                    if murder then
                        local p = Players:FindFirstChild(murder)
                        if p then
                            setRoleLabel(p, "murder")
                            if espEnabled then applyESP(p) end
                            WindUI:Notify({ Title = "⚠ MURDER FOUND", Content = murder .. " is the Murderer!", Duration = 6, Icon = "alert-triangle" })
                        end
                    end
                end

                -- Sheriff changed
                if sheriff ~= detectedSheriff then
                    if detectedSheriff then
                        local oldPlayer = Players:FindFirstChild(detectedSheriff)
                        if oldPlayer then setRoleLabel(oldPlayer, nil) end
                        if espEnabled and oldPlayer then applyESP(oldPlayer) end
                    end
                    detectedSheriff = sheriff
                    if sheriff then
                        local p = Players:FindFirstChild(sheriff)
                        if p then
                            setRoleLabel(p, "sheriff")
                            if espEnabled then applyESP(p) end
                            WindUI:Notify({ Title = "🔵 SHERIFF FOUND", Content = sheriff .. " is the Sheriff!", Duration = 6, Icon = "shield" })
                        end
                    end
                end
            end
            task.wait(0.5)
        end
    end)
end

local function stopRoleDetect()
    roleDetectEnabled = false
    if roleThread then task.cancel(roleThread) roleThread = nil end
    clearAllRoleLabels()
end

-- Clear labels when round resets (detect when no knife/gun exists anymore = new round)
task.spawn(function()
    while true do
        task.wait(2)
        if not roleDetectEnabled then continue end
        local ok, m, s = pcall(scanRoles)
        if ok and not m and not s and (detectedMurderer or detectedSheriff) then
            -- Nobody has Knife/Gun anymore = round probably reset
            clearAllRoleLabels()
            WindUI:Notify({ Title = "Round Reset", Content = "Roles cleared — new round detected", Duration = 3, Icon = "refresh-cw" })
        end
    end
end)
