    pcall(function()

-- ========================
-- Role Label System
-- ========================
local function removeRoleLabel(playerName)
    if roleLabels[playerName] then
        pcall(function() roleLabels[playerName]:Destroy() end)
        roleLabels[playerName] = nil
    end
end

local function setRoleLabel(player, roleType)
    if not player then return end
    local head = getHead(player)
    if not head then
        task.spawn(function()
            task.wait(1.5)
            head = getHead(player)
            if head then setRoleLabel(player, roleType) end
        end)
        return
    end

    removeRoleLabel(player.Name)
    if not roleType then return end

    local billboard = Instance.new("BillboardGui")
    billboard.Name        = "AAS_Role_" .. player.Name
    billboard.Size        = UDim2.fromOffset(160, 50)
    billboard.StudsOffset = Vector3.new(0, 3.5, 0)
    billboard.AlwaysOnTop = true
    billboard.Adornee     = head
    billboard.Parent      = labelGui

    local label = Instance.new("TextLabel")
    label.Size                   = UDim2.fromScale(1, 1)
    label.BackgroundTransparency = 1
    label.TextScaled             = true
    label.Font                   = Enum.Font.GothamBold
    label.TextStrokeTransparency = 0
    label.TextStrokeColor3       = Color3.fromRGB(0, 0, 0)
    label.Text       = roleType == "murder" and "⚠ MURDER" or "🔵 SHERIFF"
    label.TextColor3 = roleType == "murder"
        and Color3.fromRGB(255, 30, 30)
        or  Color3.fromRGB(50, 140, 255)
    label.Parent = billboard
    roleLabels[player.Name] = billboard
end

local function clearAllRoleLabels()
    for name in pairs(roleLabels) do removeRoleLabel(name) end
    roleLabels = {}
    latchedMurderer = nil
    latchedSheriff  = nil
    updateStatusParagraphs()
end

-- ========================
-- ESP System
-- ========================
local ESP_COLORS = {
    murder  = Color3.fromRGB(255, 50,  50),
    sheriff = Color3.fromRGB(50,  120, 255),
    default = Color3.fromRGB(200, 200, 200),
}

local function removeESP(player)
    if espHighlights[player.Name] then
        pcall(function() espHighlights[player.Name]:Destroy() end)
        espHighlights[player.Name] = nil
    end
end

local function applyESP(player)
    if player == LocalPlayer then return end
    removeESP(player)
    local char = player.Character
    if not char then return end

    local hl = Instance.new("Highlight")
    hl.Name             = "AAS_HL_" .. player.Name
    hl.FillTransparency = 0.5

    local col = ESP_COLORS.default
    if player.Name == latchedMurderer then col = ESP_COLORS.murder
    elseif player.Name == latchedSheriff then col = ESP_COLORS.sheriff end

    hl.FillColor    = col
    hl.OutlineColor = col
    hl.Adornee      = char
    hl.Parent       = espGui
    espHighlights[player.Name] = hl
end

local function refreshAllESP()
    if not espEnabled then
        for _, h in pairs(espHighlights) do pcall(function() h:Destroy() end) end
        espHighlights = {}
        return
    end
    for _, p in ipairs(Players:GetPlayers()) do applyESP(p) end
end

local function startESP()
    espEnabled = true
    refreshAllESP()

    if espThread then task.cancel(espThread) end
    espThread = task.spawn(function()
        while espEnabled do
            for _, p in ipairs(Players:GetPlayers()) do
                if p == LocalPlayer then continue end
                if not espHighlights[p.Name] or not espHighlights[p.Name].Parent then
                    applyESP(p)
                end
                -- Update color if role changed
                if espHighlights[p.Name] then
                    local col = ESP_COLORS.default
                    if p.Name == latchedMurderer then col = ESP_COLORS.murder
                    elseif p.Name == latchedSheriff then col = ESP_COLORS.sheriff end
                    pcall(function()
                        espHighlights[p.Name].FillColor    = col
                        espHighlights[p.Name].OutlineColor = col
                    end)
                end
            end
            task.wait(0.5)
        end
    end)

    for _, p in ipairs(Players:GetPlayers()) do
        p.CharacterAdded:Connect(function()
            task.wait(1)
            if espEnabled then applyESP(p) end
        end)
    end
    table.insert(espConnections, Players.PlayerAdded:Connect(function(p)
        task.wait(2)
        if espEnabled then applyESP(p) end
        p.CharacterAdded:Connect(function()
            task.wait(1)
            if espEnabled then applyESP(p) end
        end)
    end))
    table.insert(espConnections, Players.PlayerRemoving:Connect(function(p)
        removeESP(p)
    end))
end

local function stopESP()
    espEnabled = false
    for _, c in ipairs(espConnections) do pcall(function() c:Disconnect() end) end
    espConnections = {}
    if espThread then task.cancel(espThread) espThread = nil end
    for _, h in pairs(espHighlights) do pcall(function() h:Destroy() end) end
    espHighlights = {}
end

-- ========================
-- Role Detection (Latch Logic)
-- Once a player is flagged as Murder/Sheriff in a round, they keep that label
-- until the round ends — even if they drop/switch weapons.
-- ========================
local function scanForNewRoles()
    -- Returns (murdererName or nil, sheriffName or nil) for any newly detected roles
    -- (already-latched roles are ignored here; latching happens in the caller)
    local foundMurder  = nil
    local foundSheriff = nil

    for _, p in ipairs(Players:GetPlayers()) do
        if p == LocalPlayer then continue end
        local char = p.Character
        if not char then continue end

        if not foundMurder and char:FindFirstChild("Knife", true) then
            foundMurder = p.Name
        end
        if not foundSheriff and char:FindFirstChild("Gun", true) then
            foundSheriff = p.Name
        end
    end

    -- Also check workspace-level Akura_Gameza variant models
    if not foundMurder then
        for _, obj in ipairs(workspace:GetChildren()) do
            if obj.Name:find("Akura_Gameza") then
                for _, p in ipairs(Players:GetPlayers()) do
                    if p.Character and p.Character:IsAncestorOf(obj) then
                        foundMurder = p.Name break
                    end
                end
            end
        end
    end

    return foundMurder, foundSheriff
end

-- ========================
-- Round Detection
-- Heuristic: a "round" is active when >= 2 players are alive with characters.
-- Round ends when nearly everyone is dead/gone (<=1 alive) or no Knife/Gun
-- has been seen for a grace period after previously being detected.
-- ========================
local lastWeaponSeenTick = 0
local ROUND_END_GRACE    = 8 -- seconds of no weapon sighting before declaring round over

local function detectRoundActive()
    local aliveCount = 0
    for _, p in ipairs(Players:GetPlayers()) do
        local char = p.Character
        local hum  = char and char:FindFirstChildOfClass("Humanoid")
        if hum and hum.Health > 0 then aliveCount += 1 end
    end
    return aliveCount >= 2
end

local function startRoleDetect()
    roleDetectEnabled = true
    if roleThread then task.cancel(roleThread) end
    roundActive = detectRoundActive()

    roleThread = task.spawn(function()
        while roleDetectEnabled do
            local nowRoundActive = detectRoundActive()

            -- Round started
            if nowRoundActive and not roundActive then
                roundActive = true
                WindUI:Notify({ Title = "Round Started", Content = "Role scanning active", Duration = 3, Icon = "play" })
            end

            -- Scan only while round is active
            if roundActive then
                local foundMurder, foundSheriff = pcall(function() return nil, nil end)
                local ok, fm, fs = pcall(scanForNewRoles)
                if ok then
                    foundMurder, foundSheriff = fm, fs
                end

                -- Latch murderer (set once, never cleared until round ends)
                if foundMurder and not latchedMurderer then
                    latchedMurderer = foundMurder
                    local p = Players:FindFirstChild(foundMurder)
                    if p then setRoleLabel(p, "murder") end
                    updateStatusParagraphs()
                    WindUI:Notify({ Title = "⚠ MURDER FOUND", Content = foundMurder .. " is the Murderer!", Duration = 6, Icon = "alert-triangle" })
                    if espEnabled then
                        local mp = Players:FindFirstChild(foundMurder)
                        if mp then applyESP(mp) end
                    end
                end

                -- Latch sheriff (set once, never cleared until round ends)
                if foundSheriff and not latchedSheriff then
                    latchedSheriff = foundSheriff
                    local p = Players:FindFirstChild(foundSheriff)
                    if p then setRoleLabel(p, "sheriff") end
                    updateStatusParagraphs()
                    WindUI:Notify({ Title = "🔵 SHERIFF FOUND", Content = foundSheriff .. " is the Sheriff!", Duration = 6, Icon = "shield" })
                    if espEnabled then
                        local sp = Players:FindFirstChild(foundSheriff)
                        if sp then applyESP(sp) end
                    end
                end

                -- Track last time any weapon was seen
                if foundMurder or foundSheriff then
                    lastWeaponSeenTick = tick()
                end
            end

            -- Round ended: no longer enough alive players AND grace period elapsed
            if roundActive and not nowRoundActive then
                local timeSinceWeapon = tick() - lastWeaponSeenTick
                if timeSinceWeapon >= ROUND_END_GRACE then
                    roundActive = false
                    -- Clear latched roles for next round
                    clearAllRoleLabels()
                    WindUI:Notify({ Title = "Round Ended", Content = "Roles cleared — waiting for next round", Duration = 4, Icon = "refresh-cw" })
                end
            end

            task.wait(0.5)
        end
    end)
end

local function stopRoleDetect()
    roleDetectEnabled = false
    if roleThread then task.cancel(roleThread) roleThread = nil end
    clearAllRoleLabels()
end

-- ========================
-- Gun Drop Warning System
-- Scans workspace every 0.5s for a Gun model NOT inside any player's character
-- (meaning it has been dropped on the ground). Shows a big floating BillboardGui
-- label above each dropped gun, cleared when the gun is picked up or disappears.
-- ========================
local gunDropGui = Instance.new("ScreenGui")
gunDropGui.Name        = "AAS_GunDrop"
gunDropGui.ResetOnSpawn = false
pcall(function() gunDropGui.Parent = game:GetService("CoreGui") end)

local function isGunInsideCharacter(gunModel)
    for _, p in ipairs(Players:GetPlayers()) do
        local char = p.Character
        if char and (char == gunModel.Parent or char:IsAncestorOf(gunModel)) then
            return true
        end
    end
    return false
end

local function getGunDropRoot(gunModel)
    if gunModel:IsA("BasePart") then return gunModel end
    local root = gunModel:FindFirstChild("Handle")
        or gunModel:FindFirstChildWhichIsA("BasePart")
    if not root then
        for _, d in ipairs(gunModel:GetDescendants()) do
            if d:IsA("BasePart") then root = d break end
        end
    end
    return root
end

local function createGunDropLabel(gunModel)
    if gunDropLabels[gunModel] then return end
    local root = getGunDropRoot(gunModel)
    if not root then return end

    local bb = Instance.new("BillboardGui")
    bb.Name        = "AAS_GunDrop_" .. gunModel:GetFullName()
    bb.Size        = UDim2.fromOffset(220, 60)
    bb.StudsOffset = Vector3.new(0, 5, 0)
    bb.AlwaysOnTop = true
    bb.Adornee     = root
    bb.Parent      = gunDropGui

    local lbl = Instance.new("TextLabel")
    lbl.Size                   = UDim2.fromScale(1, 1)
    lbl.BackgroundColor3       = Color3.fromRGB(20, 20, 20)
    lbl.BackgroundTransparency = 0.25
    lbl.TextScaled             = true
    lbl.Font                   = Enum.Font.GothamBold
    lbl.Text                   = "🔫 GUN DROP HERE!"
    lbl.TextColor3             = Color3.fromRGB(255, 220, 0)
    lbl.TextStrokeTransparency = 0
    lbl.TextStrokeColor3       = Color3.fromRGB(0, 0, 0)
    lbl.Parent = bb

    local corner = Instance.new("UICorner")
    corner.CornerRadius = UDim.new(0, 8)
    corner.Parent = lbl

    gunDropLabels[gunModel] = bb
end

local function removeGunDropLabel(gunModel)
    if gunDropLabels[gunModel] then
        pcall(function() gunDropLabels[gunModel]:Destroy() end)
        gunDropLabels[gunModel] = nil
    end
end

local function clearAllGunDropLabels()
    for model in pairs(gunDropLabels) do removeGunDropLabel(model) end
    gunDropLabels = {}
end

local function startGunDropWarn()
    gunDropEnabled = true
    if gunDropThread then task.cancel(gunDropThread) end
    gunDropThread = task.spawn(function()
        while gunDropEnabled do
            local foundGuns = {}

            -- Scan all workspace descendants for anything named "Gun" or containing "Gun"
            -- that is NOT parented inside a player character
            for _, obj in ipairs(workspace:GetDescendants()) do
                local name = obj.Name
                if (name == "Gun" or name:find("Gun")) and not isGunInsideCharacter(obj) then
                    -- Only care about the top-level tool/model, not every BasePart called "Gun"
                    -- Accept: Tool, Model, or BasePart that is a direct child of workspace or a folder
                    local parent = obj.Parent
                    if parent == workspace or (parent and not parent:FindFirstAncestorOfClass("Model")) then
                        foundGuns[obj] = true
                        createGunDropLabel(obj)
                    elseif obj:IsA("Tool") then
                        -- Dropped tool lives directly in workspace
                        foundGuns[obj] = true
                        createGunDropLabel(obj)
                    end
                end
            end

            -- Remove labels for guns that are gone or picked up
            for model in pairs(gunDropLabels) do
                if not model.Parent or not model:IsDescendantOf(workspace) or isGunInsideCharacter(model) then
                    removeGunDropLabel(model)
                end
            end

            task.wait(0.5)
        end
    end)
end

local function stopGunDropWarn()
    gunDropEnabled = false
    if gunDropThread then task.cancel(gunDropThread) gunDropThread = nil end
    clearAllGunDropLabels()
end

-- ========================
-- Looking System
-- Every RenderStepped frame: rotate both HumanoidRootPart and Camera to face the
-- latched Murderer's HumanoidRootPart. Gracefully skips if Murder is not yet known
-- or their character is not loaded. Camera follow uses CFrame.lookAt so the player
-- always sees the murderer even through walls (pairs well with ESP).
-- ========================
local function getMurdererRoot()
    if not latchedMurderer then return nil end
    local p = Players:FindFirstChild(latchedMurderer)
    if not p or not p.Character then return nil end
    return p.Character:FindFirstChild("HumanoidRootPart")
        or p.Character:FindFirstChild("RootPart")
        or p.Character.PrimaryPart
end

local function startLooking()
    lookingEnabled = true
    if lookingConn then lookingConn:Disconnect() end
    lookingConn = RunService.RenderStepped:Connect(function()
        if not lookingEnabled then return end
        local murdererRoot = getMurdererRoot()
        if not murdererRoot then return end

        local myChar = LocalPlayer.Character
        local myHRP  = myChar and myChar:FindFirstChild("HumanoidRootPart")

        if myHRP then
            -- Rotate character body to face the murderer (only Y-axis so feet stay on ground)
            local lookPos = Vector3.new(murdererRoot.Position.X, myHRP.Position.Y, murdererRoot.Position.Z)
            pcall(function()
                myHRP.CFrame = CFrame.new(myHRP.Position, lookPos)
            end)
        end

        -- Also pivot camera toward the murderer
        pcall(function()
            Camera.CFrame = CFrame.lookAt(Camera.CFrame.Position, murdererRoot.Position)
        end)
    end)
end

local function stopLooking()
    lookingEnabled = false
    if lookingConn then lookingConn:Disconnect() lookingConn = nil end
end

-- Auto-stop Looking when Murder latch is cleared (new round)
local _origClearAllRoleLabels = clearAllRoleLabels
clearAllRoleLabels = function()
    _origClearAllRoleLabels()
    if lookingEnabled then
        -- Don't disable the toggle, just stop rotating until a new Murder is latched
        if lookingConn then lookingConn:Disconnect() lookingConn = nil end
        task.spawn(function()
            -- Restart the conn so it resumes as soon as latchedMurderer is set again
            task.wait(1)
            if lookingEnabled then startLooking() end
        end)
    end
end

-- ========================
-- Noclip System
-- ========================
local function startNoclip()
    noclipEnabled = true
    if noclipThread then task.cancel(noclipThread) end
    noclipThread = task.spawn(function()
        while noclipEnabled do
            local char = LocalPlayer.Character
            if char then
                for _, part in ipairs(char:GetDescendants()) do
                    if part:IsA("BasePart") and part.CanCollide then
                        part.CanCollide = false
                    end
                end
            end
            task.wait(0.1)
        end
        -- Restore collision when stopped
        local char = LocalPlayer.Character
        if char then
            for _, part in ipairs(char:GetDescendants()) do
                if part:IsA("BasePart") then
                    part.CanCollide = true
                end
            end
        end
    end)
end

local function stopNoclip()
    noclipEnabled = false
    if noclipThread then task.cancel(noclipThread) noclipThread = nil end
    local char = LocalPlayer.Character
    if char then
        for _, part in ipairs(char:GetDescendants()) do
            if part:IsA("BasePart") then
                pcall(function() part.CanCollide = true end)
            end
        end
    end
end

LocalPlayer.CharacterAdded:Connect(function()
    task.wait(1)
    if noclipEnabled then startNoclip() end
end)

-- Auto-Load Config on Start
-- ========================
task.spawn(function()
    task.wait(1.5)
    pcall(function()
        ConfigManager:Load("default")
        task.wait(0.3)
        if GetValue("esp", false)          then startESP()          end
        if GetValue("roleDetect", false)   then startRoleDetect()   end
        if GetValue("gunDropWarn", false)  then startGunDropWarn()  end
        if GetValue("lookingEnabled", false) then startLooking()    end
        if GetValue("noclip", false)       then startNoclip()       end
    end)
end)

-- ========================
-- Ready Notify
-- ========================
WindUI:Notify({
    Title    = "AAS : Murder Mystery Detector ✅",
    Content  = "Script loaded! Enable Role Detect and ESP to start.",
    Duration = 5,
    Icon     = "shield-alert",
})
